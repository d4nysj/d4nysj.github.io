 *    - Watchdog de seguridad: corta el riego si supera duracion maxima

String getTimestampStr() {

  struct tm ti;

  if (!getLocalTime(&ti)) return "??/??/?? ??:??:??";

  char buf[20];

  strftime(buf, sizeof(buf), "%d/%m/%y %H:%M:%S", &ti);

  return String(buf);

}



void addLog(const char* nivel, const char* msg) {

  String ts = getTimestampStr();

  LogEntry& e = logBuffer[logHead];

  strncpy(e.timestamp, ts.c_str(), 19); e.timestamp[19] = 0;

  strncpy(e.nivel,     nivel,       5);  e.nivel[5]      = 0;

  strncpy(e.msg,       msg,         79); e.msg[79]       = 0;

  logHead = (logHead + 1) % LOG_MAX;

  if (logCount < LOG_MAX) logCount++;

  Serial.printf("[%s][%s] %s\n", e.timestamp, nivel, msg);

}



void logInfo(const char* msg)  { addLog("INFO",  msg); }

void logWarn(const char* msg)  { addLog("WARN",  msg); }

void logError(const char* msg) { addLog("ERROR", msg); }



void logInfoF(const char* fmt, ...) {

  char buf[80];

  va_list args;

  va_start(args, fmt);

  vsnprintf(buf, sizeof(buf), fmt, args);

  va_end(args);

  logInfo(buf);

}

//  SENSOR DE HUMEDAD

// ═══════════════════════════════════════════════════════════════════════════════

// El sensor capacitivo tipico devuelve ~4095 en seco y ~1500 en agua

// Convertimos a % donde 0% = completamente seco, 100% = empapado



int leerHumedad() {

  if (!cfg.usarHumedad) return -1;

  int raw = analogRead(HUMEDAD_PIN);

  // Calibracion tipica: seco=3800, mojado=1500 (ajusta segun tu sensor)

  const int SECO   = 3800;

  const int MOJADO = 1500;

  int pct = map(constrain(raw, MOJADO, SECO), SECO, MOJADO, 0, 100);

  return pct;

}



// Retorna true si la humedad es suficiente y NO hay que regar

bool sueloHumedo() {

  if (!cfg.usarHumedad) return false;

  int h = leerHumedad();

  if (h < 0) return false;

  return h >= cfg.umbralSeco;

}



// ═══════════════════════════════════════════════════════════════════════════════

//  CONTROL DEL RELE

// ═══════════════════════════════════════════════════════════════════════════════



void iniciarRiego(int segundos, const char* tipo) {

  if (regandoAhora) {

    logWarn("Riego solicitado pero ya esta activo, ignorado");

    return;

  }



  // Comprobacion de seguridad: no superar MAX_DURACION

  if (segundos > MAX_DURACION) {

    logWarn("Duracion solicitada supera el maximo de seguridad, recortada");

    segundos = MAX_DURACION;

  }



  // Comprobar humedad si sensor activo

  if (cfg.usarHumedad && sueloHumedo()) {

    int h = leerHumedad();

    logInfoF("Riego %s cancelado: suelo humedo al %d%%", tipo, h);

    return;

  }



  digitalWrite(RELAY_PIN, RELAY_ON);

  regandoAhora = true;

  riegoInicio  = millis();

  riegoDurMs   = (unsigned long)segundos * 1000UL;

  riegoTipo    = String(tipo);



  logInfoF("Riego %s iniciado: %ds (%.1f L estimados)",

           tipo, segundos, segundos * cfg.litrosPorSeg);

}



void detenerRiego() {

  if (!regandoAhora) return;



  digitalWrite(RELAY_PIN, RELAY_OFF);



  // Calcular duracion real y litros

  unsigned long durReal = (millis() - riegoInicio) / 1000;

  float litros = durReal * cfg.litrosPorSeg;



  // Guardar en historial

  RiegoRecord& rec = historial[histHead];

  String ts = getTimestampStr();

  strncpy(rec.timestamp, ts.c_str(), 19); rec.timestamp[19] = 0;

  strncpy(rec.tipo, riegoTipo.c_str(), 9); rec.tipo[9] = 0;

  rec.segundos = durReal;

  rec.litros   = litros;

  histHead = (histHead + 1) % HIST_MAX;

  if (histCount < HIST_MAX) histCount++;



  // Estadisticas acumuladas

  totalRiegos++;

  totalLitros += litros;

  guardarEstadisticas();



  regandoAhora = false;

  logInfoF("Riego finalizado: %lus reales, %.2f L (total acum: %lu riegos, %.1f L)",

           durReal, litros, totalRiegos, totalLitros);

}

void conectarWifi() {

  logInfo("Iniciando conexion WiFi (red oculta WPA2)...");

  WiFi.mode(WIFI_STA);

  WiFi.begin(WIFI_SSID, WIFI_PASS);

  int intentos = 0;

  while (WiFi.status() != WL_CONNECTED && intentos < 40) {

    delay(500);

    intentos++;

  }

  if (WiFi.status() == WL_CONNECTED) {

    wifiConectado = true;

    char buf[60];

    snprintf(buf, sizeof(buf), "WiFi conectado. IP: %s  RSSI: %d dBm",

             WiFi.localIP().toString().c_str(), WiFi.RSSI());

    logInfo(buf);

  } else {

    wifiConectado = false;

    logError("WiFi: no se pudo conectar. Reintentando en 30s...");

  }

}



void checkWifi() {

  if (millis() - ultimoWifiCheck < WIFI_RETRY_MS) return;

  ultimoWifiCheck = millis();

  if (WiFi.status() != WL_CONNECTED) {

    wifiConectado = false;

    wifiReconexiones++;

    char buf[60];

    snprintf(buf, sizeof(buf), "WiFi perdido. Reconexion #%d...", wifiReconexiones);

    logWarn(buf);

    WiFi.disconnect();

    delay(200);

    WiFi.begin(WIFI_SSID, WIFI_PASS);

    // No bloqueamos el loop, se comprobara en el siguiente ciclo

  } else if (!wifiConectado) {

    wifiConectado = true;

    // Reconectado: resincronizar NTP

    configTzTime(TIMEZONE, NTP_SERVER);

    char buf[60];

    snprintf(buf, sizeof(buf), "WiFi reconectado. IP: %s", WiFi.localIP().toString().c_str());

    logInfo(buf);

  }

}

String uptime() {

  unsigned long seg = millis() / 1000;

  int d = seg / 86400; seg %= 86400;

  int h = seg / 3600;  seg %= 3600;

  int m = seg / 60;    seg %= 60;

  char buf[40];

  snprintf(buf, sizeof(buf), "%dd %02dh %02dm %02ds", d, h, m, (int)seg);

  return String(buf);

}

  return "#94a3b8";

}

//  LOOP

// ═══════════════════════════════════════════════════════════════════════════════

void loop() {

  server.handleClient();



  // Watchdog: cortar riego si supera tiempo maximo

  if (regandoAhora && millis() - riegoInicio >= riegoDurMs) {

    detenerRiego();

  }



  // Watchdog extra de seguridad: nunca mas de MAX_DURACION segundos

  if (regandoAhora && millis() - riegoInicio > (unsigned long)MAX_DURACION * 1000UL) {

    logError("WATCHDOG: riego activo mas de MAX_DURACION. Forzando parada de emergencia.");

    digitalWrite(RELAY_PIN, RELAY_OFF);

    regandoAhora = false;

  }



  // WiFi watchdog / reconexion automatica

  checkWifi();



  // Scheduler (cada segundo)

  static unsigned long ultimaCheck = 0;

  if (millis() - ultimaCheck < 1000) return;

  ultimaCheck = millis();



  struct tm ti;

  if (!getLocalTime(&ti)) return;



  int ahora = ti.tm_hour * 60 + ti.tm_min;



  // Riego 1

  if (cfg.activo1) {

    int prog1 = cfg.hora1 * 60 + cfg.min1;

    if (ahora == prog1 && ti.tm_sec == 0 && ultimaMinRiego1 != ahora) {

      logInfoF("Riego AUTO1 programado (%02d:%02d)", cfg.hora1, cfg.min1);

      iniciarRiego(cfg.duracion, "AUTO1");

      ultimaMinRiego1 = ahora;

    }

  }

  if (ahora != (cfg.hora1 * 60 + cfg.min1)) ultimaMinRiego1 = -1;



  // Riego 2

  if (cfg.activo2) {

    int prog2 = cfg.hora2 * 60 + cfg.min2;

    if (ahora == prog2 && ti.tm_sec == 0 && ultimaMinRiego2 != ahora) {

      logInfoF("Riego AUTO2 programado (%02d:%02d)", cfg.hora2, cfg.min2);

      iniciarRiego(cfg.duracion, "AUTO2");

      ultimaMinRiego2 = ahora;

    }

  }

  if (ahora != (cfg.hora2 * 60 + cfg.min2)) ultimaMinRiego2 = -1;

}

Mantener un huerto en perfectas condiciones exige constancia, sobre todo cuando llega el verano y los riegos no perdonan un solo descuido. Después de varios prototipos, he consolidado la versión definitiva de mi sistema para mantener todo verde y optimizado: el **Riego Automático ESP32 v3.0 "NASA Edition"**.

Quería huir de los típicos temporizadores de grifo comerciales que son cerrados y limitados. La idea era diseñar un hardware y firmware a medida que no solo regara a sus horas, sino que tomara decisiones basándose en la humedad real de la tierra y me permitiera auditar todo desde el teléfono.

![Dashboard Web del ESP32](https://github.com/d4nysj/Esp32_Riego_Automatico/blob/main/Images/1.png?raw=true)

### ¿Qué hace a esta versión tan especial?

He programado el firmware en C++ estructurando el código de forma robusta para evitar sorpresas. Estas son las características clave que le he implementado:

* **Sincronización NTP y Zona Horaria Autónoma**: Se conecta a `pool.ntp.org` y aplica automáticamente el horario de Madrid (CET/CEST). Olvídate de ajustar la hora manualmente en los cambios de estación.
* **Decisión basada en Datos**: Mediante un sensor capacitivo analógico conectado al `GPIO34`, el ESP32 evalúa si la tierra ya está lo suficientemente húmeda. Si es así, cancela inteligentemente el riego programado para ahorrar agua.
* **Persistencia NVS Flash**: Usando la librería `Preferences`, el ESP32 guarda toda la configuración, así como el conteo total de riegos y los litros estimados que ha bombeado. Si hay un corte de luz, el sistema se levanta y retoma su estado anterior sin perder un solo dato.
* **Seguridad Anti-inundaciones**: Implementé un doble *Watchdog*. Si por algún bug o desconexión el relé (conectado al `GPIO26`) se queda encendido más de los segundos estipulados de seguridad, el sistema corta la corriente forzosamente.

### Interfaz Minimalista Integrada

Todo el panel de control está embebido directamente en la memoria del microcontrolador. Sin depender de nubes externas ni suscripciones. Usando HTML, CSS puro y un poco de JS asíncrono, he diseñado una web en formato "Dark Mode" estilo terminal de comandos.

![Pestaña de Diagnóstico y Logs](https://github.com/d4nysj/Esp32_Riego_Automatico/blob/main/Images/2.png?raw=true)

Desde esta interfaz se pueden configurar las horas de activación, forzar un riego manual, consultar los **logs del sistema** (guardados en un buffer circular en RAM) y ver las métricas de hardware como la temperatura de la CPU, uso de RAM y calidad de la señal WiFi.
